struct BIT {                                   // 1-based
    int n; vector<ll> t;
    BIT(int n) : n(n), t(n + 1) {}
    void add(int i, ll v) { for (; i <= n; i += i & -i) t[i] += v; }
    ll sum(int i) { ll s = 0; for (; i > 0; i -= i & -i) s += t[i]; return s; }
    int kth(ll k) {                            // smallest i with sum(i) >= k (counts >= 0)
        int pos = 0;
        for (int pw = 1 << __lg(n); pw; pw >>= 1)
            if (pos + pw <= n && t[pos + pw] < k) pos += pw, k -= t[pos];
        return pos + 1;
    }
};
struct RangeBIT {                              // range add, range sum (1-based)
    BIT b1, b2;
    RangeBIT(int n) : b1(n + 1), b2(n + 1) {}
    void add(int l, int r, ll v) { b1.add(l, v); b1.add(r + 1, -v); b2.add(l, v * (l - 1)); b2.add(r + 1, -v * r); }
    ll pre(int i) { return b1.sum(i) * i - b2.sum(i); }
    ll sum(int l, int r) { return pre(r) - pre(l - 1); }
};
struct SegTree {                               // iterative: point set, query [l, r), 0-based
    int n; vector<ll> t; ll ID = LLONG_MAX;    // ID = identity of merge
    ll merge(ll a, ll b) { return min(a, b); }
    SegTree(int n) : n(n), t(2 * n, LLONG_MAX) {}
    void set(int i, ll v) { for (t[i += n] = v; i > 1; i >>= 1) t[i >> 1] = merge(t[i & ~1], t[i | 1]); }
    ll query(int l, int r) {
        ll L = ID, R = ID;                     // keep the sides apart: merge may not commute
        for (l += n, r += n; l < r; l >>= 1, r >>= 1) {
            if (l & 1) L = merge(L, t[l++]);
            if (r & 1) R = merge(t[--r], R);
        }
        return merge(L, R);
    }
};
struct LazySeg {                               // range add; {sum, min} on [l, r], 0-based
    int n; vector<ll> sm, mn, lz;
    LazySeg(int n) : n(n), sm(4 * n), mn(4 * n), lz(4 * n) {}
    void apply(int x, int len, ll v) { sm[x] += v * len; mn[x] += v; lz[x] += v; }
    void push(int x, int l, int r) {
        if (!lz[x]) return;
        int m = (l + r) / 2;
        apply(2 * x, m - l + 1, lz[x]); apply(2 * x + 1, r - m, lz[x]); lz[x] = 0;
    }
    void add(int ql, int qr, ll v, int x = 1, int l = 0, int r = -1) {
        if (r < 0) r = n - 1;
        if (qr < l || r < ql) return;
        if (ql <= l && r <= qr) { apply(x, r - l + 1, v); return; }
        push(x, l, r); int m = (l + r) / 2;
        add(ql, qr, v, 2 * x, l, m); add(ql, qr, v, 2 * x + 1, m + 1, r);
        sm[x] = sm[2 * x] + sm[2 * x + 1]; mn[x] = min(mn[2 * x], mn[2 * x + 1]);
    }
    pair<ll, ll> query(int ql, int qr, int x = 1, int l = 0, int r = -1) {
        if (r < 0) r = n - 1;
        if (qr < l || r < ql) return {0, LLONG_MAX};
        if (ql <= l && r <= qr) return {sm[x], mn[x]};
        push(x, l, r); int m = (l + r) / 2;
        auto a = query(ql, qr, 2 * x, l, m), b = query(ql, qr, 2 * x + 1, m + 1, r);
        return {a.first + b.first, min(a.second, b.second)};
    }
};
vector<int> mx;                                // recursive max tree, size 4n, node x covers [l, r]
int firstAtLeast(int k, int x, int l, int r) { // descend: leftmost a[i] >= k, or -1
    if (mx[x] < k) return -1;
    if (l == r) return l;
    int m = (l + r) / 2, L = firstAtLeast(k, 2 * x, l, m);
    return L != -1 ? L : firstAtLeast(k, 2 * x + 1, m + 1, r);
}   // LC 3479 fruits [4,2,5], baskets [3,5,4] → 1 unplaced
